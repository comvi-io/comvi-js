<Steps>

1. **Compose the host**

     ```ts title="src/i18n.ts"
     import { createI18n } from '@comvi/react';
     import { loader } from '@comvi/core/loader';

     export const i18n = createI18n({ locale: 'en' }).use(loader());
     ```

2. **Register a catalog**

     ```ts title="src/i18n.ts"
     i18n.registerLoader({ uk: () => import('./locales/uk.json') });
     ```

3. **A second host, composed nowhere**

     ```ts title="src/other.ts"
     other.registerLoader({ uk: () => import('./locales/uk.json') });
     ```

</Steps>
